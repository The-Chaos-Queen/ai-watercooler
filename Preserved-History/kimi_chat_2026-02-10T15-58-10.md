## Architecture

- **Mechanism**: A Python class `ProstheticHand` that wraps safe movement, clicking, and typing.
- **Safety**:
    - `pyautogui.FAILSAFE = True` (Default)
    - `pyautogui.PAUSE = 1.0` (Slows down actions to 1 sec per step for observability)

### 2. The Eye (Perception Layer)
- **Tool**: `pyautogui.screenshot()`
- **Processing**: Local VLM (Llava v1.6 or Moondream) via LM Studio (or Ollama) API.
- **Workflow**:
    1.  Script takes screenshot.
    2.  Sends image + query ("Where is the 'File' menu?") to Local VLM.
    3.  VLM returns coordinates/description.
    4.  Script converts VLM output to `(x, y)` targets.

### 3. The Brain (Coordination Layer)
- **Role**: C.H.E.E.S.E. (Cloud) provides the high-level strategy ("Open VS Code and create a new file").
- **Execution**: The local script executes the tactic loop (See -> Move -> Click -> Verify).

"""
eye_core.py — The Prosthetic Eye
--------------------------------
Hybrid perception: VLM for semantics, OpenCV for precision.
Falls back to grid-based exploration when both fail.
"""

import base64
import io
import json
import sqlite3
from dataclasses import dataclass
from enum import Enum, auto
from pathlib import Path
from typing import Optional, Tuple, List
import cv2
import numpy as np
import requests
from PIL import Image


class Certainty(Enum):
    PRECISE = auto()      # OpenCV template match > threshold
    APPROXIMATE = auto()  # VLM gave region/description
    EXPLORATORY = auto()  # Grid search fallback
    FAILED = auto()       # No target found


@dataclass
class VisualTarget:
    x: int
    y: int
    certainty: Certainty
    confidence: float  # 0.0-1.0
    source: str        # "template_match", "vlm_parse", "grid_search"
    description: str   # What we were looking for
    
    def __repr__(self):
        return f"Target({self.x}, {self.y}, {self.certainty.name}, {self.confidence:.2f})"


class VisualLibrary:
    """
    Persistent storage of learned UI element signatures.
    Templates are stored as hashes, not raw images, for privacy.
    """
    
    def __init__(self, db_path: str = "visual_memory.db"):
        self.db_path = db_path
        self._init_db()
        # In-memory cache of loaded templates
        self._template_cache: dict[str, np.ndarray] = {}
    
    def _init_db(self):
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                CREATE TABLE IF NOT EXISTS elements (
                    id INTEGER PRIMARY KEY,
                    semantic_name TEXT UNIQUE,  -- "VS Code File Menu"
                    context_hash TEXT,           -- Hash of surrounding UI state
                    template_path TEXT,          -- Path to .png template
                    roi_x, roi_y, roi_w, roi_h,  -- Region of interest hint
                    success_count INTEGER DEFAULT 0,
                    last_used TIMESTAMP
                )
            """)
    
    def recall(self, semantic_name: str, current_context: Image.Image) -> Optional[Tuple[np.ndarray, Tuple[int, int, int, int]]]:
        """
        Retrieve template + ROI hint if we've seen this element before
        in similar context.
        """
        # TODO: Context similarity matching (perceptual hash of full screenshot)
        with sqlite3.connect(self.db_path) as conn:
            row = conn.execute(
                "SELECT template_path, roi_x, roi_y, roi_w, roi_h FROM elements WHERE semantic_name = ?",
                (semantic_name,)
            ).fetchone()
            
        if not row:
            return None
            
        template_path, x, y, w, h = row
        
        if semantic_name not in self._template_cache:
            template = cv2.imread(template_path, cv2.IMREAD_GRAYSCALE)
            if template is None:
                return None
            self._template_cache[semantic_name] = template
            
        return self._template_cache[semantic_name], (x, y, w, h)
    
    def memorize(self, semantic_name: str, screenshot: Image.Image, 
                 bbox: Tuple[int, int, int, int], context: str):
        """Store successful recognition for future use."""
        x, y, w, h = bbox
        template = screenshot.crop((x, y, x+w, y+h))
        
        safe_name = semantic_name.replace(" ", "_").lower()
        template_path = f"templates/{safe_name}_{hash(context) % 10000}.png"
        Path("templates").mkdir(exist_ok=True)
        template.save(template_path)
        
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                INSERT OR REPLACE INTO elements 
                (semantic_name, context_hash, template_path, roi_x, roi_y, roi_w, roi_h, last_used)
                VALUES (?, ?, ?, ?, ?, ?, ?, datetime('now'))
            """, (semantic_name, str(hash(context)), template_path, x, y, w, h))


class ProstheticEye:
    """
    Main interface. Coordinates between semantic understanding (VLM)
    and geometric precision (OpenCV).
    """
    
    def __init__(self, vlm_endpoint: str = "http://localhost:1234/v1/chat/completions",
                 vlm_model: str = "llava-v1.6"):
        self.vlm_endpoint = vlm_endpoint
        self.vlm_model = vlm_model
        self.library = VisualLibrary()
        
        # Multi-scale template matching parameters
        self.match_threshold = 0.8
        self.scales = [0.8, 0.9, 1.0, 1.1, 1.2]  # Handle DPI/resolution variance
        
    def _encode_image(self, image: Image.Image) -> str:
        """Base64 encode for VLM API."""
        buffer = io.BytesIO()
        image.save(buffer, format="PNG")
        return base64.b64encode(buffer.getvalue()).decode()
    
    def _query_vlm_semantic(self, image: Image.Image, task_description: str) -> dict:
        """
        Ask VLM: What type of UI element? What's nearby? 
        NOT: Where exactly is it? (VLMs are bad at pixels)
        """
        prompt = f"""Analyze this screenshot for: {task_description}

Respond in JSON:
{{
    "element_type": "button|menu|icon|text_field|unknown",
    "visual_description": "color, shape, text label if readable",
    "relative_location": "top-left|top-center|top-right|center|bottom...",
    "nearby_landmarks": ["text labels or icons near the target"],
    "confidence": 0.0-1.0
}}"""

        response = requests.post(
            self.vlm_endpoint,
            json={
                "model": self.vlm_model,
                "messages": [{
                    "role": "user",
                    "content": [
                        {"type": "text", "text": prompt},
                        {"type": "image_url", "image_url": {"url": f"data:image/png;base64,{self._encode_image(image)}"}}
                    ]
                }],
                "temperature": 0.1  # Deterministic for UI analysis
            },
            timeout=30
        )
        
        try:
            content = response.json()["choices"][0]["message"]["content"]
            # Handle possible markdown code blocks
            if "```json" in content:
                content = content.split("```json")[1].split("```")[0]
            return json.loads(content)
        except (KeyError, json.JSONDecodeError) as e:
            return {"element_type": "unknown", "confidence": 0.0, "error": str(e)}
    
    def _template_match_adaptive(self, screenshot: np.ndarray, 
                                  template: np.ndarray,
                                  roi: Optional[Tuple[int, int, int, int]] = None
                                  ) -> Optional[Tuple[int, int, float]]:
        """
        Multi-scale template matching within optional region of interest.
        Returns (x, y, confidence) or None.
        """
        if roi:
            rx, ry, rw, rh = roi
            search_area = screenshot[ry:ry+rh, rx:rx+rw]
            offset_x, offset_y = rx, ry
        else:
            search_area = screenshot
            offset_x, offset_y = 0, 0
            
        best_match = None
        best_confidence = 0
        
        for scale in self.scales:
            resized = cv2.resize(template, None, fx=scale, fy=scale)
            if resized.shape[0] > search_area.shape[0] or resized.shape[1] > search_area.shape[1]:
                continue
                
            result = cv2.matchTemplate(search_area, resized, cv2.TM_CCOEFF_NORMED)
            _, max_val, _, max_loc = cv2.minMaxLoc(result)
            
            if max_val > best_confidence:
                best_confidence = max_val
                best_match = (max_loc[0] + offset_x, max_loc[1] + offset_y)
                
        if best_confidence >= self.match_threshold:
            return (*best_match, best_confidence)
        return None
    
    def _exploratory_grid(self, screenshot: np.ndarray, 
                          semantic_hint: dict) -> Optional[VisualTarget]:
        """
        Last resort: Systematic grid search based on VLM's relative location hint.
        Divides screen into regions, prioritizes based on semantic hint.
        """
        h, w = screenshot.shape[:2]
        
        # Priority regions based on VLM hint
        regions = {
            "top-left": [(0, 0, w//2, h//2)],
            "top-center": [(w//4, 0, w//2, h//2)],
            "top-right": [(w//2, 0, w//2, h//2)],
            "center": [(w//3, h//3, w//3, h//3)],
            # ... etc
        }
        
        hint_location = semantic_hint.get("relative_location", "center")
        priority = regions.get(hint_location, regions["center"])
        
        # Look for high-contrast edges (likely interactive elements)
        gray = cv2.cvtColor(screenshot, cv2.COLOR_BGR2GRAY) if len(screenshot.shape) == 3 else screenshot
        edges = cv2.Canny(gray, 50, 150)
        
        for x, y, rw, rh in priority:
            roi_edges = edges[y:y+rh, x:x+rw]
            # Find contours that might be buttons
            contours, _ = cv2.findContours(roi_edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
            
            for cnt in contours:
                if cv2.contourArea(cnt) > 100:  # Minimum size filter
                    cx, cy, cw, ch = cv2.boundingRect(cnt)
                    # Return center of first plausible target
                    return VisualTarget(
                        x=x+cx+cw//2, y=y+cy+ch//2,
                        certainty=Certainty.EXPLORATORY,
                        confidence=0.4,
                        source="grid_search",
                        description=semantic_hint.get("visual_description", "unknown")
                    )
        return None
    
    def locate(self, description: str, screenshot: Image.Image) -> VisualTarget:
        """
        Main entry point. Cascades through precision levels.
        """
        screenshot_cv = cv2.cvtColor(np.array(screenshot), cv2.COLOR_RGB2BGR)
        
        # LEVEL 1: Muscle Memory (fastest, most precise)
        memory = self.library.recall(description, screenshot)
        if memory:
            template, roi = memory
            match = self._template_match_adaptive(screenshot_cv, template, roi)
            if match:
                x, y, conf = match
                return VisualTarget(x, y, Certainty.PRECISE, conf, "template_match", description)
        
        # LEVEL 2: Semantic Understanding + Fresh Template Match
        semantic = self._query_vlm_semantic(screenshot, description)
        
        if semantic.get("confidence", 0) < 0.3:
            return VisualTarget(0, 0, Certainty.FAILED, 0.0, "none", description)
        
        # If VLM described it well, try to find by visual features
        # (In practice, you'd have a library of generic templates per element_type)
        generic_template = self._synthesize_template(semantic)
        if generic_template is not None:
            match = self._template_match_adaptive(screenshot_cv, generic_template)
            if match:
                x, y, conf = match
                # Store for next time
                self.library.memorize(description, screenshot, (x-20, y-10, 40, 20), str(semantic))
                return VisualTarget(x, y, Certainty.APPROXIMATE, conf, "vlm_guided_match", description)
        
        # LEVEL 3: Exploratory Search
        exploratory = self._exploratory_grid(screenshot_cv, semantic)
        if exploratory:
            return exploratory
            
        return VisualTarget(0, 0, Certainty.FAILED, 0.0, "none", description)

    def verify_action(self, before: Image.Image, after: Image.Image, 
                      expected_change: str) -> Tuple[bool, float]:
        """
        Post-action verification. Did the click produce expected result?
        """
        # Simple structural similarity
        before_arr = np.array(before)
        after_arr = np.array(after)
        
        if before_arr.shape != after_arr.shape:
            return True, 1.0  # Window changed size, likely dialog opened

    def click_target(self, target: VisualTarget, force: bool = False):
        if target.certainty == Certainty.FAILED and not force:
            raise SafetyError(f"Refusing to click failed target: {target.description}")
            
        if target.certainty == Certainty.EXPLORATORY and not force:
            # Require confirmation for exploratory clicks

        # Execute with verification
        before = pyautogui.screenshot()

        success, confidence = self.eye.verify_action(before, after, f"clicked {target.description}")
        if not success:
